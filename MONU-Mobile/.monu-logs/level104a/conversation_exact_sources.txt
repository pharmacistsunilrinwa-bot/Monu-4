     3	data class MONUConversationState(
     4	    val messages: List<MONUConversationMessage> = emptyList(),
     5	    val isProcessing: Boolean = false,
     6	    val lastError: String? = null,
     7	    val restored: Boolean = false
     8	)
     9	
    10	class MONUConversationStateController(
    11	    private val repository: MONUConversationRepository =
    12	        MONULocalConversationRepository(),
    13	    private val persistence: MONUConversationPersistence =
    14	        MONULocalConversationSnapshotStore()
    15	) {
    16	
    17	    private var currentState =
    18	        MONUConversationState()
    19	
    20	    fun state(): MONUConversationState {
    21	        return currentState
    22	    }
    23	
    24	    fun beginProcessing() {
    25	        currentState =
    26	            currentState.copy(
    27	                isProcessing = true,
    28	                lastError = null
    29	            )
    30	    }
    31	
    32	    fun finishProcessing() {
    33	        currentState =
    34	            currentState.copy(
    35	                isProcessing = false
    36	            )
    37	
    38	        persist()
    39	    }
    40	
    41	    fun addUserMessage(
    42	        content: String
    43	    ): MONUConversationMessage {
    44	
    45	        val message =
    46	            repository.append(
    47	                role = MONUConversationRole.USER,
    48	                content = content
    49	            )
    50	
    51	        refreshMessages()
    52	        persist()
    53	
    54	        return message
    55	    }
    56	
    57	    fun addAssistantMessage(
    58	        content: String
    59	    ): MONUConversationMessage {
    60	
    61	        val message =
    62	            repository.append(
    63	                role = MONUConversationRole.ASSISTANT,
    64	                content = content
    65	            )
    66	
    67	        refreshMessages()
    68	        persist()
    69	
    70	        return message
    71	    }
    72	
    73	    fun setError(
    74	        message: String
    75	    ) {
    76	        currentState =
    77	            currentState.copy(
    78	                isProcessing = false,
    79	                lastError = message
    80	            )
    81	
    82	        persist()
    83	    }
    84	
    85	    fun restoreConversation() {
    86	        val snapshot = persistence.restore()
    87	
    88	        currentState =
    89	            if (snapshot == null) {
    90	                currentState.copy(
    91	                    restored = true
    92	                )
    93	            } else {
    94	                MONUConversationState(
    95	                    messages = snapshot.messages,
    96	                    isProcessing = false,
    97	                    lastError = null,
    98	                    restored = true
    99	                )
   100	            }
   101	    }
   102	
   103	    fun clearConversation() {
   104	        repository.clear()
   105	        persistence.clear()
   106	
   107	        currentState =
   108	            MONUConversationState(
   109	                restored = true
   110	            )
   111	    }
   112	
   113	    fun snapshot(): MONUConversationSnapshot {
   114	        return MONUConversationSnapshot(
   115	            messages = currentState.messages,
   116	            updatedAt = System.currentTimeMillis()
   117	        )
   118	    }
   119	
   120	    private fun refreshMessages() {
   121	        currentState =
   122	            currentState.copy(
   123	                messages = repository.messages()
   124	            )
   125	    }
   126	
   127	    private fun persist() {
   128	        persistence.save(snapshot())
   129	    }
   130	}

    36	    }

     3	import com.monu.mobile.feature.offline.MONUOfflineCommandContract
     4	import com.monu.mobile.feature.offline.MONUOfflineCommandRequest
     5	import com.monu.mobile.feature.offline.MONUOfflineCommandResponse
     6	import com.monu.mobile.feature.offline.MONUOfflineCommandRouter
     7	
     8	data class MONUConversationExecutionResult(
     9	    val userMessage: MONUConversationMessage,
    10	    val assistantMessage: MONUConversationMessage,
    11	    val commandResponse: MONUOfflineCommandResponse
    12	)
    13	
    14	class MONUOfflineConversationBridge(
    15	    private val commandContract: MONUOfflineCommandContract =
    16	        MONUOfflineCommandRouter(),
    17	    private val stateController: MONUConversationStateController =
    18	        MONUConversationStateController()
    19	) {
    20	
    21	    fun conversationState(): MONUConversationState {
    22	        return stateController.state()
    23	    }
    24	
    25	    fun execute(
    26	        command: String
    27	    ): MONUConversationExecutionResult {
    28	
    29	        stateController.beginProcessing()
    30	
    31	        return try {
    32	
    33	            val userMessage =
    34	                stateController.addUserMessage(
    35	                    content = command
    36	                )
    37	
    38	            val response =
    39	                commandContract.execute(
    40	                    MONUOfflineCommandRequest(
    41	                        command = command
    42	                    )
    43	                )
    44	
    45	            val assistantMessage =
    46	                stateController.addAssistantMessage(
    47	                    content = response.response
    48	                )
    49	
    50	            stateController.finishProcessing()
    51	
    52	            MONUConversationExecutionResult(
    53	                userMessage = userMessage,
    54	                assistantMessage = assistantMessage,
    55	                commandResponse = response
    56	            )
    57	
    58	        } catch (error: Exception) {
    59	
    60	            val safeMessage =
    61	                error.message
    62	                    ?: "MONU could not process the command locally."
    63	
    64	            stateController.setError(safeMessage)
    65	
    66	            val userMessage =
    67	                stateController.addUserMessage(
    68	                    content = command
    69	                )
    70	
    71	            val assistantMessage =
    72	                stateController.addAssistantMessage(
    73	                    content = safeMessage
    74	                )
    75	
    76	            MONUConversationExecutionResult(
    77	                userMessage = userMessage,
    78	                assistantMessage = assistantMessage,
    79	                commandResponse =
    80	                    MONUOfflineCommandResponse(
    81	                        handled = false,
    82	                        intent =
    83	                            com.monu.mobile.feature.offline
    84	                                .MONUOfflineCommandIntent.UNKNOWN,
    85	                        response = safeMessage
    86	                    )
    87	            )
    88	        }
    89	    }
    90	
    91	    fun clearConversation() {
    92	        stateController.clearConversation()
    93	    }
    94	}
